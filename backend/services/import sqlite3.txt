import sqlite3
import json

DB_PATH = "backend/races.db"


def save_pre_race_result(race_id: str, summary: dict):
    conn = sqlite3.connect(DB_PATH)
    cur = conn.cursor()

    cur.execute(
        """
        UPDATE races
        SET pre_race_run = 1,
            pre_race_summary = ?
        WHERE race_id = ?
        """,
        (json.dumps(summary, ensure_ascii=False), race_id),
    )

    conn.commit()
    conn.close()
